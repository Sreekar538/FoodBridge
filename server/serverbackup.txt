
const express = require("express");

const app = express();
const PORT = 5000;

app.use(express.json());

// Temporary donation storage for testing.
// Data resets when the server restarts.
let donations = [];

// Home route
app.get("/", function (req, res) {
    res.json({
        message: "FoodBridge backend is running successfully!"
    });
});

// Get all donations
app.get("/api/donations", function (req, res) {
    res.json({
        success: true,
        count: donations.length,
        donations: donations
    });
});

// Add a donation
app.post("/api/donations", function (req, res) {
    const { foodName, quantity, description } = req.body;

    if (!foodName || !quantity || !description) {
        return res.status(400).json({
            success: false,
            message: "Food name, quantity, and description are required."
        });
    }

    const donation = {
        id: Date.now().toString(),
        foodName: foodName,
        quantity: Number(quantity),
        description: description,
        status: "Available",
        createdAt: new Date().toISOString()
    };

    donations.push(donation);

    res.status(201).json({
        success: true,
        message: "Donation added successfully!",
        donation: donation
    });
});

// Start server
app.listen(PORT, function () {
    console.log(`FoodBridge server running at http://localhost:${PORT}`);
});
